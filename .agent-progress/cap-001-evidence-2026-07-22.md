- Environment: `HIPFIRE_DETERMINISTIC=1`, `HIPFIRE_EMIT_TOKEN_IDS=1`,
  `HIPFIRE_DFLASH_DRAFT=""`.

Run `EP=1` and requested `EP=2` in separate fresh daemon processes. Parse only
the `tok_id` member of each `committed` JSON envelope; `t_ms` is deliberately
excluded because it is timing data, not model output.

```bash
python3 - <<'PY'
import hashlib, json, os, subprocess

model = os.path.expanduser("~/.hipfire/models/qwen3.5-9b.mq4")
daemon = "./target/release/examples/daemon"
env = os.environ.copy()
env.update({
    "HIPFIRE_DETERMINISTIC": "1",
    "HIPFIRE_EMIT_TOKEN_IDS": "1",
    "HIPFIRE_DFLASH_DRAFT": "",
})

def run(ep):
    params = {"max_seq": 512}
    if ep != 1:
        params["ep"] = ep
    requests = (
        {"type": "load", "model": model, "params": params},
        {"type": "generate", "id": "cap001-parity",
         "prompt": "The capital of France is", "temperature": 0.0,
         "max_tokens": 32},
        {"type": "unload"},
    )
    proc = subprocess.run(
        [daemon], input="\n".join(map(json.dumps, requests)) + "\n",
        text=True, capture_output=True, env=env, timeout=540, check=True,
    )
    return [json.loads(line)["tok_id"] for line in proc.stdout.splitlines()
            if "tok_id" in line]

ep1, ep2 = run(1), run(2)
assert ep1 == ep2, (ep1, ep2)
print(len(ep1), hashlib.sha256(json.dumps(ep1).encode()).hexdigest())
PY
```
